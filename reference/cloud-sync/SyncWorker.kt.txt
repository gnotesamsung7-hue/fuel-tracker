package com.parengmark.fueltracker.work

import android.content.Context
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.parengmark.fueltracker.FuelTrackerApp
import com.parengmark.fueltracker.data.sync.FirebaseSyncRepository

class SyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        val database = (applicationContext as FuelTrackerApp).database
        return try {
            FirebaseSyncRepository(database).sync()
            Result.success()
        } catch (e: Exception) {
            Result.retry()
        }
    }
}
